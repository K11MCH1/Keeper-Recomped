cmake_minimum_required(VERSION 3.22)
project(keeper_original_probe LANGUAGES C CXX)
if(NOT SR_SOURCE)
    message(FATAL_ERROR "Set SR_SOURCE to the disposable SR source copy")
endif()
file(GLOB SR_C CONFIGURE_DEPENDS "${SR_SOURCE}/*.c" "${SR_SOURCE}/udis86-1.6/libudis86/*.c")
file(GLOB SR_CPP CONFIGURE_DEPENDS "${SR_SOURCE}/*.cpp")
add_executable(SR ${SR_C} ${SR_CPP})
target_include_directories(SR PRIVATE "${SR_SOURCE}/udis86-1.6")
target_compile_features(SR PRIVATE cxx_std_20)
if(MSVC)
    target_compile_definitions(SR PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
